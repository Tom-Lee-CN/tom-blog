# 代码块演示

本文用于演示博客中各种编程语言代码块的渲染效果，包含 JavaScript、HTML、CSS、Vue 和 Rust 的示例。

## JavaScript

```js
// 计算斐波那契数列
function fibonacci(n) {
  if (n <= 1) return n
  let a = 0
  let b = 1
  for (let i = 2; i <= n; i++) {
    const temp = a + b
    a = b
    b = temp
  }
  return b
}

// 使用 async/await 处理异步
async function fetchData(url) {
  try {
    const response = await fetch(url)
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`)
    }
    const data = await response.json()
    return data
  } catch (error) {
    console.error('请求失败：', error)
    return null
  }
}

// 数组操作示例
const numbers = [1, 2, 3, 4, 5]
const doubled = numbers.map((n) => n * 2)
const evenNumbers = numbers.filter((n) => n % 2 === 0)
const sum = numbers.reduce((acc, cur) => acc + cur, 0)
```

## Vue

```vue
<template>
  <div class="counter">
    <p>当前计数：{{ count }}</p>
    <button type="button" @click="increment">增加</button>
    <button type="button" @click="decrement">减少</button>
  </div>
</template>

<script setup>
import { ref, computed } from 'vue'

// 计数器状态
const count = ref(0)

// 派生状态：是否为偶数
const isEven = computed(() => count.value % 2 === 0)

// 操作方法
function increment() {
  count.value++
}

function decrement() {
  count.value--
}
</script>

<style scoped>
.counter {
  display: flex;
```

## Rust

```rust
// 定义一个简单的结构体
struct Rectangle {
    width: u32,
    height: u32,
}

impl Rectangle {
    // 关联函数：构造函数
    fn new(width: u32, height: u32) -> Self {
        Self { width, height }
    }

    // 方法：计算面积
    fn area(&self) -> u32 {
        self.width * self.height
    }
}

// 主函数
fn main() {
    let rect = Rectangle::new(10, 20);
    println!("矩形的面积是：{}", rect.area());

    // 使用迭代器和闭包
    let numbers = vec![1, 2, 3, 4, 5];
    let sum: u32 = numbers.iter().map(|x| x * 2).sum();
    println!("双倍后的总和：{}", sum);
}
```
